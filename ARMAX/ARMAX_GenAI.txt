import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from datetime import datetime

class ARMAX_NN:
    def __init__(self, n_inputs, n_outputs, n_delayed_inputs, n_delayed_outputs, 
                 n_delayed_noise, n_hidden, n_horizon, activation=np.tanh):
        # Network Parameters
        self.n_inputs = n_inputs
        self.n_outputs = n_outputs
        self.n_delayed_inputs = n_delayed_inputs
        self.n_delayed_outputs = n_delayed_outputs
        self.n_delayed_noise = n_delayed_noise
        self.n_hidden = n_hidden
        self.n_horizon = n_horizon
        self.activation = activation
        
        # Initialize weights and biases
        self.weights = {
            "input_hidden": np.random.randn(n_inputs + n_delayed_inputs + n_delayed_outputs + n_delayed_noise, n_hidden),
            "hidden_output": np.random.randn(n_hidden, n_outputs * n_horizon),
        }
        self.biases = {
            "hidden": np.random.randn(n_hidden),
            "output": np.random.randn(n_outputs * n_horizon),
        }
    
    # Activation Function and its Derivative
    def activation_derivative(self, x):
        return 1 - np.tanh(x) ** 2

    # Loss Function
    def loss(self, e, gain=1, gamma=2):
        return np.power(gain, np.tanh(gamma * e)) * np.log(np.cosh(e))

    def loss_(self, e, gain=1, gamma=2):
        return np.power(gain, np.tanh(gamma * e)) * np.tanh(e)

    # Forward Pass
    def forward(self, x):
        self.hidden = self.activation(np.dot(x, self.weights["input_hidden"]) + self.biases["hidden"])
        self.output = np.dot(self.hidden, self.weights["hidden_output"]) + self.biases["output"]
        return self.output

    # Backward Pass
    def backward(self, x, y, lr, momentum, prev_deltas):
        error = self.output - y
        output_grad = self.loss_(error).reshape(-1, self.n_outputs * self.n_horizon)
        hidden_grad = np.dot(output_grad, self.weights["hidden_output"].T) * self.activation_derivative(self.hidden)
        
        deltas = {
            "hidden_output": momentum * prev_deltas["hidden_output"] - lr * np.dot(self.hidden.T, output_grad),
            "input_hidden": momentum * prev_deltas["input_hidden"] - lr * np.dot(x.T, hidden_grad),
            "output_bias": momentum * prev_deltas["output_bias"] - lr * np.sum(output_grad, axis=0),
            "hidden_bias": momentum * prev_deltas["hidden_bias"] - lr * np.sum(hidden_grad, axis=0),
        }
        
        # Update weights and biases
        self.weights["hidden_output"] += deltas["hidden_output"]
        self.weights["input_hidden"] += deltas["input_hidden"]
        self.biases["output"] += deltas["output_bias"]
        self.biases["hidden"] += deltas["hidden_bias"]
        
        return deltas

    # Save Network Topology and Weights
    def save_network(self, filename):
        with open(filename, 'w') as f:
            f.write(f"Topology: {self.n_inputs}-{self.n_hidden}-{self.n_outputs}\n")
            f.write(f"Weights: {self.weights}\n")
            f.write(f"Biases: {self.biases}\n")

    # Load Network Topology and Weights
    def load_network(self, filename):
        with open(filename, 'r') as f:
            lines = f.readlines()
            self.weights = eval(lines[1].split(": ")[1])
            self.biases = eval(lines[2].split(": ")[1])

# Training Function
def train_network(model, training_file, n_epochs=1000, n_show=100):
    # Load data
    data = pd.read_csv(training_file)
    time = data.iloc[:, 0]
    inputs = data.iloc[:, 1:model.n_inputs + 1].values
    outputs = data.iloc[:, model.n_inputs + 1:].values

    losses = []
    prev_deltas = {key: np.zeros_like(value) for key, value in model.weights.items()}
    prev_deltas.update({key: np.zeros_like(value) for key, value in model.biases.items()})

    for epoch in range(1, n_epochs + 1):
        y_pred = model.forward(inputs)
        error = y_pred - outputs
        loss = np.mean(model.loss(error))
        losses.append(loss)

        model.backward(inputs, outputs, lr=0.01, momentum=0.9, prev_deltas=prev_deltas)

        # Plot every n_show epochs
        if epoch % n_show == 0:
            plt.figure()
            plt.plot(time, outputs.flatten(), 'k.', label="Training Data")
            plt.plot(time, y_pred.flatten(), 'r', label="Predicted Outputs")
            plt.title(f"Epoch {epoch}")
            plt.legend()
            plt.show()

    # Save loss plot
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    plt.figure()
    plt.plot(range(1, n_epochs + 1), losses, label="Loss")
    plt.xlabel("Epochs")
    plt.ylabel("Loss")
    plt.title("Loss vs. Epochs")
    plt.legend()
    plt.savefig(f"Loss_Plot@{timestamp}.pdf")
    plt.savefig(f"Loss_Plot@{timestamp}.jpeg")

    return losses

# Testing Function
def test_network(model, testing_file, n_horizon, n_skip=5):
    # Load test data
    data = pd.read_csv(testing_file)
    time = data.iloc[:, 0]
    inputs = data.iloc[:, 1:model.n_inputs + 1].values
    outputs = data.iloc[:, model.n_inputs + 1:].values

    predictions = []
    for t_k in range(0, len(inputs) - n_horizon, n_skip):
        input_data = inputs[t_k:t_k + n_horizon]
        forecast = model.forward(input_data)
        predictions.append(forecast)

        plt.figure()
        plt.plot(time[:t_k + n_horizon], outputs[:t_k + n_horizon].flatten(), 'k.', label="Historic Data")
        plt.plot(time[t_k:t_k + n_horizon], forecast.flatten(), 'r', label="Forecast")
        plt.fill_between(time[t_k:t_k + n_horizon], forecast.flatten() - 0.1, forecast.flatten() + 0.1, alpha=0.2)
        plt.title(f"Forecast at t = {t_k}")
        plt.legend()
        plt.show()

    # Save predictions as CSV
    pd.DataFrame(predictions).to_csv("Simulation_Results.csv", index=False)

# Example Usage
if __name__ == "__main__":
    model = ARMAX_NN(n_inputs=2, n_outputs=1, n_delayed_inputs=1, n_delayed_outputs=1, 
                     n_delayed_noise=1, n_hidden=10, n_horizon=3)
    train_network(model, "training_file.csv")
    model.save_network("network.net")
    model.load_network("network.net")
    test_network(model, "testing_file.csv", n_horizon=3)

Lets create a five layer recurrent neural network class, called DyNet, to learn nonlinear dynamic systems (NLS). The NLS has input vector, u(k) of dimension n_inputs (default=1), output vector, y(k) of dimension n_outputs (default=1), and state vector, x(k) of dimension n_states (default=2).
We are going to model NLS with state equations:
x(k+1) = f(x(k),u(k),w(k)); 
y(k) = h(x(k),u(k),v(k)), 
where w(k) is a process noise vector of dimension n_process_noise (default=n_states), and v(k) is a sensor noise vector of dimension n_sensor_noise (default=n_outputs).  

The DyNet topology is as follows: 

The first layer has n_inputs+n_states+nodes+n_process_noise nodes with 'linear' activation function.  The first n_inputs nodes take the DyNet inputs, u, whether from a data file for training, training_file (default training_file='training_data.csv') in CSV format or from a sream provided to a simulate() method as a Pandas dataframe.  The next n_states nodes are fed back from the third layer with time delay (more later).  The last n_process_noise nodes represent the process noise, w(k), that is generated when needed, from a Gaussian Normal distribution with zero meand and process_stddev standard deviation (default = 0.02).

The training_file should have as first colum the time (k), the following n_inputs columns are the inputs, u, from u[1], u[2], ..., u[n_inputs], the next n_outputs columns are the putputs, y, from y[1], y[2], ..., y[n_outputs].  In theory, there might be additional n_states columns that are the states, x, from x[1], x[2], ..., x[n_states], if they are available-which is not the case in general.

The second layer has n_hidden_states nodes (default=5), that may have biases (default is hidden_state_bias=True) and the default activation function is 'tanh()',  Both biases and activation functions should be selectable as well.  

The third layer has n_states+n_sensor_noise nodes, that have 'linear' activation function and have biases as default.  The third layer n_states nodes are connected to the fourth layer and are also fed back to the n_states nodes in the input layer, with a time delay.  Basically, these first three layers can realize the state dynamics: x(k+1) = f(x(k),u(k),w(k)).  The rest of the third layer nodes, n_sensor_noise, represent the sensor noise, v(k), have no inputs and only connect to the fourth layer.  Their values are generated from a Gaussian Normal distribution with zero meand and sensors_stddev standard deviation (default = 0.01).  The fed back links from the third n_states nodes to the first layer n_states nodes, have a value of '1', are fix (they don't adapt) although you can backpropagate through them, and the connect 1:1, not fully connected.  That is the first only to the first, the second only to the second, and so on.

The fourth and fifth layers approximate the sensors.  The fourth layer takes inputs from the third layer, including the states, x(k+1), sensor noise, v(k+1) and also from the first layer n_inputs nodes, u(k) with a delay, since the third layer states are x(k+1).  The fourth layer has n_hidden_sensors nodes (default=3) with default activation function 'tanh()' and biases as default.

The last layer has n_outputs nodes, have 'linear' activation function and have biases (as default).  These nodes represent the outputs, y(k) and are the network outputs.
